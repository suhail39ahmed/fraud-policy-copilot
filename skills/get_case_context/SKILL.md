-- params: :TXN_ID
USE DATABASE FRAUD_COPILOT;
USE SCHEMA DEMO;

WITH seed AS (
  SELECT * FROM TRANSACTIONS WHERE TXN_ID = :TXN_ID
)
SELECT
  OBJECT_CONSTRUCT(
    'txn', OBJECT_CONSTRUCT(*),
    'customer', (
      SELECT OBJECT_CONSTRUCT(*)
      FROM CUSTOMERS c
      WHERE c.CUSTOMER_ID = (SELECT CUSTOMER_ID FROM seed)
    ),
    'recent_txns', (
      SELECT ARRAY_AGG(OBJECT_CONSTRUCT(*)) WITHIN GROUP (ORDER BY TXN_TS DESC)
      FROM TRANSACTIONS t
      WHERE t.CUSTOMER_ID = (SELECT CUSTOMER_ID FROM seed)
        AND t.TXN_TS <= (SELECT TXN_TS FROM seed)
    )
  ) AS CASE_CONTEXT
FROM seed;
